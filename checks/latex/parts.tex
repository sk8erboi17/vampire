Here a set of parts is the set of all the parts of a set, which first-order
logic cannot say: the mathematical engine goes first, and Vampire is asked
only when the engine has no answer, for a proof from the laws of those sets.
The last two statements quantify over the parts of a set: there Vampire
looks for a refutation of the clauses that the independent checker builds
from the statement, with the parts that the engine proposes. The last two
suppose two sets equipotent: the engine proves the second; the first it does
not, and Vampire proves it with the bijection given as two maps inverse to
each other.
\begin{enumerate}
\item $\displaystyle S \neq \emptyset \implies (\exists! X \in \mathcal{P}(S)\ \forall x \in S\ (x \in X))$
\item $\displaystyle \forall x \in S\ \exists X \in \mathcal{P}(S)\ (x \in X)$
\item $\displaystyle \forall X \in \mathcal{P}(S)\ \exists x \in S\ (x \in X)$
\item $\displaystyle \exists S\ \exists F\ ((F \subseteq \mathcal{P}(S) \land \emptyset \in F) \land \forall X \in F\ \exists x \in S\ (x \in X))$
\item $\displaystyle G_{1} = \{(0, 0), (1, 1), (0, 1), (1, 0)\} \implies (\forall x, y\ ((x, y) \in G_{1} \implies (y, x) \in G_{1}))$
\item $\displaystyle G_{1} = \{(1, 2), (2, 3)\} \implies (\forall x, y, z\ (((x, y) \in G_{1} \land (y, z) \in G_{1}) \implies (x, z) \in G_{1}))$
\item $\displaystyle (z \in S \land s \colon S \to S \land f \colon S \to S \land (\forall X \subseteq S\ ((z \in X \land \forall y \in S\ (y \in X \implies s(y) \in X)) \implies \forall y \in S\ (y \in X))) \land f(z)=z \land (\forall y \in S\ (f(s(y))=s(f(y))))) \implies \forall y \in S\ (f(y)=y)$
\item $\displaystyle \exists S \neq \emptyset\ \exists f \colon S \to S\ (\forall X \subseteq S\ (f(X) \neq X))$
\item $\displaystyle (S \sim T \land \forall x, y, z, t \in S\ (x = y \lor x = z \lor x = t \lor y = z \lor y = t \lor z = t)) \implies \forall x, y, z, t \in T\ (x = y \lor x = z \lor x = t \lor y = z \lor y = t \lor z = t)$
\item $\displaystyle (S \sim T \land \forall x, y \in S\ (x = y)) \implies \forall u, v \in T\ (u = v)$
\end{enumerate}
