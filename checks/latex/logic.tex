\title{Statements of first-order logic on arbitrary sets}

\maketitle
Statements about elements, maps and relations of sets that are not given:
the frontend unfolds them and Vampire proves or refutes them. The sets may
be empty.
\begin{enumerate}
\item $\displaystyle (f \colon S \to S \land \forall x \in S\ (f(f(x)) = x)) \implies \forall x, y \in S\ (f(x) = f(y) \implies x = y)$
\item $\displaystyle (f \colon S \to T \land g \colon T \to S \land \forall x \in S\ (g(f(x)) = x)) \implies \forall y \in T\ \exists x \in S\ (f(x) = y)$
\item $\displaystyle (f \colon S \to S \land \forall x, y \in S\ (f(x) = f(y) \implies x = y)) \implies \forall y \in S\ \exists x \in S\ (f(x) = y)$
\item $\displaystyle (u \in S \land X \subseteq S) \implies u \in X$
\item $\displaystyle \forall f \in S^{S}\ \exists x \in S\ (f(x) = f(x))$
\item $\displaystyle (f \colon S \to T \land g \colon T \to S \land \forall x \in S\ (g(f(x)) = x)) \implies \forall x, y \in S\ (f(x) = f(y) \implies x = y)$
\item $\displaystyle (R \subseteq S \times S \land \forall x \in S\ ((x, x) \in R)) \implies (R = \emptyset)$
\item $\displaystyle (R \subseteq S \times S \land \forall x, y \in S\ ((x, y) \in R \implies (y, x) \in R) \land \forall x, y, z \in S\ (((x, y) \in R \land (y, z) \in R) \implies (x, z) \in R) \land \forall x \in S\ \exists y \in S\ ((x, y) \in R)) \implies \forall x \in S\ ((x, x) \in R)$
\item $\displaystyle (f \colon S \to T \land |S| \ge 2) \implies (|T| \ge 2)$
\item $\displaystyle (X \subseteq S \land Y \subseteq S \land |X| = 2 \land |Y| = 1 \land X \cap Y = \emptyset) \implies (|X \cup Y| = 3)$
\item $\displaystyle X \subseteq S \implies (|X| = 1 \iff \exists a \in S\ (X = \{a\}))$
\end{enumerate}
